# 坐标系说明:
#   NED:  North-East-Down, 飞控原生坐标系, 起飞点为原点
#   Field: 场地坐标系 (forward, right, height), 以起飞点+场地航向为基准
#   interface.field_to_ned(forward, right, height) → PositionNedYaw(north, east, down, yaw)
#
#   关键位置:
#     起飞点:        field(0, 0, 0)
#     投放区中心:    field(DROP_ZONE_DISTANCE_M≈30, 0, ?)
#     侦察区起点:    field(-3.5, 59.0, 5.0)
#     侦察区终点:    field(+3.5, 56.0, 5.0)
#
# =============================================================================
# 一、FSM 全局调度
# =============================================================================

MissionFSM.run():
    build_mission()   # 构建任务队列
    connect_and_setup()
    arm_and_offboard()

    for each state in mission_queue:
        state.enter(interface)       ← 进入状态

        while True:                  ← 执行循环 (每个 tick)
            if not global_guard_check():  # 健康检查 (GPS/电量/连接)
                RTL()                     # 不健康 → 返航降落
                return

            done, _ = state.execute(interface)
            if done: break
            sleep(1/20)              ← 50ms 一个 tick

        state.exit(interface)        ← 退出状态

        # 读取路由标记, 决定下一状态
        phase = interface.shared.get("next_phase")
        if phase == "recon":     → 插入 ReconState
        if phase == "rectangle": → 插入矩形搜索状态
        # (实际实现可能用动态队列或 Router 状态)

# =============================================================================
# 二、矩形搜索中断逻辑 (写入 shared, 与 align_precise 联动)
# =============================================================================

矩形搜索状态:
    无人机沿矩形航线飞行, 视觉持续扫描圆柱体

    当检测到圆柱体:
        1. 保存中断点:
           interface.shared["position_left"] = (forward, right)
           # ↑ 无人机当前场地的 (前, 右) 坐标, 投放后要飞回这里

        2. 保存目标信息:
           interface.shared["drop_targets"] = [target_15cm, target_20cm]
           # ↑ 两个圆柱体的粗略 NED 偏移, 供 AlignState 使用

        3. 中断矩形航线 → FSM 切换到 AlignState

# =============================================================================
# 三、AlignState (align.py) —— 粗对准
# =============================================================================

AlignState.enter():
    # 从 shared 读取目标
    targets = interface.shared.get("drop_targets")
    target  = targets[0]  # 或 1, 取决于 bottle_index

    # 飞向目标圆柱体上方 ~2.5m 高度
    sp = field_to_ned(
        DROP_ZONE_DISTANCE_M + target.ned_offset[0],
        target.ned_offset[1],
        DROP_ALIGN_ALTITUDE_M  # 2.5m
    )
    update_setpoint(sp)

AlignState.execute() 循环:
    # 子阶段 1: 等待下降到位
    if alt <= 3.0:
        phase = "servo"

    # 子阶段 2: 视觉伺服粗对准
    if phase == "servo":
        # 检测圆柱体
        cylinders = detector.detect_cylinders(frame, alt)
        best = match_target(cylinders)  # 找最近的
        offset_x, offset_y = best.ned_offset

        # 对准判据: 偏移 < 0.05m
        if abs(offset_x) < 0.05 and abs(offset_y) < 0.05:
            # 标记对准完成 → FSM 切到 AlignPreciseState
            interface.shared["bottle_1_aligned"] = True
            return True

        # 未对准 → P 控制调整
        sp = field_to_ned(
            DROP_ZONE_DISTANCE_M + offset_x * VISUAL_SERVO_KP,
            offset_y * VISUAL_SERVO_KP,
            alt
        )
        update_setpoint(sp)

    return False

# =============================================================================
# 四、AlignPreciseState (align_precise.py) —— 速度伺服精细对准 + 投放 + 返航
# =============================================================================
#
# 与位置伺服的区别:
#   位置: dx → 算新位置 → set_position_ned → PX4位置环 → 移动
#   速度: dx → 算速度  → set_velocity_ned → PX4速度环 → 移动 (少一层, 响应更快)
#
# 心跳支持: interface._velocity_mode 决定发位置还是发速度 (20Hz)
#   update_setpoint(sp) → 切回位置模式
#   update_velocity(v)  → 切速度模式

AlignPreciseState.__init__(bottle_index):
    self.bottle_index = bottle_index
    self.timeout_s    = 120

# --- 可调参数 ---
# 速度伺服
ALIGN_TIMEOUT = 30.0           # 超时强制投放
ANGULAR_VEL_THRESHOLD = 0.5    # 角速度低于此值→稳定
DISTANCE_THRESHOLD = 0.05      # 圆筒中心距视觉中心 < 此值→对准
CYCLE_INTERVAL = 0.2           # 视觉检测间隔
KP_VEL = 2.0                   # 速度增益: 偏移(m)×KP=速度(m/s)
MAX_VEL = 1.0                  # 水平速度上限 (m/s)
# 投放后导航
ASCEND_COMPLETE_MARGIN = 0.5
POST_HOVER_TIME = 1.0

# --- 进入状态 (每轮投放调用一次) ---
AlignPreciseState.enter(interface):
    self._drop_done = False

    # 对准节拍变量 (首 tick 才初始化)
    self._align_initialized = False
    self._align_start = 0.0
    self._last_check_time = 0.0
    self._detected_cylinder = None

    # 导航子步骤: 0=上升 1=返航 2=悬停 3=分支
    self._nav_step = 0
    self._nav_step_start = 0.0
    self._nav_step0_init = True
    self._nav_step1_init = True

    # 上升锚点: 投放时 _do_drop 用 odometry 更新
    self._ascent_north = interface._last_setpoint.north_m
    self._ascent_east  = interface._last_setpoint.east_m

# --- 每 tick 调用 (不阻塞 FSM) ---
AlignPreciseState.execute(interface):
    if self._drop_done == False:
        self._drop_done = self._align_tick(interface)
        return False, None
    else:
        done, _ = self._post_drop_navigation(interface)
        if done: self.is_completed = True
        return done, None

# ============================================================================
# _align_tick — 单节拍速度伺服
# ============================================================================

AlignPreciseState._align_tick(interface):
    now = time.monotonic()

    # -- 首 tick 初始化 --
    if not self._align_initialized:
        self._align_initialized = True
        self._align_start = now
        self._last_check_time = 0.0
        self._goal = interface.shared.setdefault("goal", [0, 0])
        # ↑ setdefault: 第二轮不覆盖已有 [1,0]
        self._detected_cylinder = None

    # -- 节流: < 0.2s 跳过 --
    if now - self._last_check_time < CYCLE_INTERVAL:
        return False
    self._last_check_time = now

    elapsed = now - self._align_start

    # -- 超时: 先发零速停住再投放 (防止旧速度残留漂移) --
    if elapsed > ALIGN_TIMEOUT:
        interface.update_velocity(VelocityNedYaw(0, 0, 0, yaw))
        self._do_drop(interface)
        return True

    # -- 1. 机体角速度 (合成幅值) --
    try:
        rate = 读取三轴角速度
        angular_vel = √(roll² + pitch² + yaw²)
    except:
        return False

    # -- 2. 视觉检测 --
    try:
        alt      = interface.get_altitude()
        detector = get_detector()
        frame    = _capture_frame_async()
        cylinders = detector.detect_cylinders(frame, alt)
    except:
        return False

    if 没检测到:
        # 发零速停住, 等待目标重新出现
        interface.update_velocity(VelocityNedYaw(0, 0, 0, yaw))
        return False

    # 取距离摄像头中心最近的圆柱体
    self._detected_cylinder = min(
        cylinders,
        key=lambda c: √(c.ned_offset[0]² + c.ned_offset[1]²)
    )
    dx, dy   = cylinder.ned_offset       # NED 偏移 (m)
    distance = √(dx² + dy²)

    # -- 3. 对准判据 --
    if distance < DISTANCE_THRESHOLD AND angular_vel < ANGULAR_VEL_THRESHOLD:
        # 发零速悬停, 然后投放
        interface.update_velocity(VelocityNedYaw(0, 0, 0, yaw))
        self._do_drop(interface)
        return True

    # -- 4. 未对准: 速度伺服 --
    # 偏移越大速度越快, 偏移→0 速度→0, 无需累加
    v_north = dx * KP_VEL              # 0.2m偏移 → 0.4m/s
    v_east  = dy * KP_VEL

    # 限速
    speed = √(v_north² + v_east²)
    if speed > MAX_VEL:
        v_north *= MAX_VEL / speed
        v_east  *= MAX_VEL / speed

    interface.update_velocity(
        VelocityNedYaw(v_north, v_east, 0.0, yaw))
    # ↑ 心跳循环 20Hz 持续发送, PX4 速度环直接驱动电机
    return False

# --- 投放 + 善后 ---
AlignPreciseState._do_drop(interface):
    # 1. 舵机投放
    drop = DropState(bottle_index=self.bottle_index)
    drop.execute(interface)                  # 舵机 1.0 → 0.5s → -1.0

    # 2. 根据检测直径更新 goal
    if self._detected_cylinder != None:
        dia = cylinder.diameter
        if   |dia - 0.15| < 0.02:  goal[0] = 1   # 15cm 完成
        elif |dia - 0.20| < 0.02:  goal[1] = 1   # 20cm 完成
        else:                      不更新

    # 3. 上升锚点: 从 odometry 读投放时实际位置
    pos = interface.get_position_ned()
    self._ascent_north = pos.north_m
    self._ascent_east  = pos.east_m

# --- 阶段 B, 投放后导航 (子步骤状态机) ---
AlignPreciseState._post_drop_navigation(interface):
    goal     = interface.shared.get("goal", [0,0])
    pos_left = interface.shared.get("position_left")  # 矩形中断点

    # ================================================================
    # Step 0: 垂直上升至巡航高度 (CRUISE_ALTITUDE_M = 7.0m)
    # ================================================================
    if self._nav_step == 0:

        if 首次进入本步骤:
            self._nav_step0_init = False
            记录开始时间
            # 保持投放后水平位置, 仅改高度
            # update_setpoint 会同时关闭速度模式, 恢复位置控制
            sp = PositionNedYaw(
                self._ascent_north, self._ascent_east,
                -CRUISE_ALTITUDE_M, interface.FIELD_YAW_DEG
            )
            update_setpoint(sp)
            print("上升至 7m")

        alt = 当前高度
        if |alt - CRUISE_ALTITUDE_M| < ASCEND_COMPLETE_MARGIN:
            self._nav_step = 1
            记录本步骤开始时间

        return False, None

    # ================================================================
    # Step 1: 飞回矩形航线中断点
    # ================================================================
    if self._nav_step == 1:

        if pos_left == None:
            print("警告: position_left 未保存")
            self._nav_step = 2
            return False, None

        if 首次进入本步骤:
            self._nav_step1_init = False
            记录开始时间
            sp = field_to_ned(pos_left[0], pos_left[1], CRUISE_ALTITUDE_M)
            update_setpoint(sp)
            print("飞回中断点")

        # 距离估算: 从上升锚点 (NED) → pos_left (NED)
        pos_left_ned = field_to_ned(pos_left[0], pos_left[1], CRUISE_ALTITUDE_M)
        dist = √((pos_left_ned.north - self._ascent_north)² +
                  (pos_left_ned.east  - self._ascent_east)²)
        est_time = dist / TRANSIT_SPEED_MPS

        # 主判据: 水平距离 < 0.5m + 高度到位
        当前NED = get_position_ned()
        h_dist = √((pos_left_ned.north - 当前NED.north)² +
                    (pos_left_ned.east  - 当前NED.east)²)

        if h_dist < 0.5 AND |alt - CRUISE_ALTITUDE_M| < ASCEND_COMPLETE_MARGIN:
            self._nav_step = 2
            return False, None

        # 兜底: 1.5倍预计时间
        if 已过时间 > est_time * 1.5:
            self._nav_step = 2

        return False, None

    # ================================================================
    # Step 2: 悬停 1s 稳定
    # ================================================================
    if self._nav_step == 2:
        # 不发布新设定值, 无人机在中断点原地悬停
        if 已过时间 > 1.0:
            self._nav_step = 3           # → 推进到 Step 3
        return False, None

    # ================================================================
    # Step 3: 分支路由
    # ================================================================
    if self._nav_step == 3:
        if goal == [1, 1]:
            # 两个圆筒都完成 → 侦察
            interface.shared["next_phase"] = "recon"
            print("路由 → recon")
        else:
            # 还有未完成的 → 继续矩形搜索
            interface.shared["next_phase"] = "rectangle"
            print("路由 → rectangle")

        return True, None                # 状态结束

    return True, None                    # fallback

# 五、ReconState (recon.py) —— 双线侦察扫描
# =============================================================================

ReconState.enter(interface):
    #
    # 生成 4 个航点
    #
    fwd  = 59.0                          # 侦察区前向距离
    half = 3.5                           # 半宽 (留 0.5m 余量)
    alt  = 5.0                           # 扫描高度 (单线覆盖 ~5.8m)
    gap  = 3.0                           # 两线前向间距

    航点 = [
        (-3.5, 59.0, 5.0),              # WP0: 线1起点 (也是 transit 目标)
        (+3.5, 59.0, 5.0),              # WP1: 线1终点
        (+3.5, 56.0, 5.0),              # WP2: 线2起点
        (-3.5, 56.0, 5.0),              # WP3: 线2终点
    ]

    阶段 = "transit"                     # transit → scan → stabilize

    # 发布 WP0 设定值, 无人机飞向扫描起点
    update_setpoint(field_to_ned(-3.5, 59.0, 5.0))

    # 估算 transit 距离
    当前位置 = get_position_ned()
    WP0_NED   = field_to_ned(-3.5, 59.0, 5.0)
    transit_dist = √((WP0_NED.north - 当前.north)² +
                      (WP0_NED.east  - 当前.east)²)
    self._transit_est = transit_dist / 3.0  # 预计时间

ReconState.execute(interface):

    # -- 阶段 1: 飞向扫描起点 --
    if 阶段 == "transit":
        alt = 当前高度
        WP0_NED = field_to_ned(-3.5, 59.0, 5.0)
        当前 = get_position_ned()
        h_dist = √((WP0_NED.north - 当前.north)² +
                    (WP0_NED.east  - 当前.east)²)

        if h_dist < 1.0 AND |alt - 5.0| < 0.5:   # 主判据
            阶段 = "scan"
        elif 已过时间 > transit_est * 1.5:         # 兜底
            阶段 = "scan"
        return False

    # -- 阶段 2: 航点扫描 --
    if 阶段 == "scan":
        航段距离 = 上一航点 → 当前航点的距离
        预计时间 = 航段距离 / 3.0

        if 本段已过时间 > 预计时间 AND |alt - 5.0| < 0.5:
            self._current_wp += 1

            if 还有下一航点:
                update_setpoint(field_to_ned(下一航点))
                重置本段计时
            else:
                阶段 = "stabilize"
        return False

    # -- 阶段 3: 扫描后稳定 --
    if 阶段 == "stabilize":
        if 已过时间 > 1.0:
            return True                 # → FSM 切到 Land

    return False

# 六、PrecisionLandState (land.py) —— 降落
# =============================================================================

PrecisionLandState.enter(interface):
    alt = 当前高度
    sp = field_to_ned(0, 0, alt)         # 飞回起飞点上方
    update_setpoint(sp)
    interface.land()                     # 切 PX4 Land 模式

PrecisionLandState.execute(interface):
    检查 armed 状态
    if not armed:                        # PX4 着陆后自动 disarm
        return True                      # 任务结束
